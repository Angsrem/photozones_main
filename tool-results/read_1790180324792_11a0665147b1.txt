    74→function slugify(s: string) {
    75→  return (s || '')
    76→    .toLowerCase()
    77→    .replace(/[^a-z0-9а-я]+/gi, '-')
    78→    .replace(/^-+|-+$/g, '')
    79→    .slice(0, 80);
    80→}

   345→
   346→function PhotozoneEditor({
   347→  photozone,
   348→  onClose,
   349→  onSaved,
   350→}: {
   351→  photozone: PhotoZone | null;
   352→  onClose: () => void;
   353→  onSaved: () => void;
   354→}) {
   355→  const isEdit = !!photozone;
   356→  const [title, setTitle] = React.useState(photozone?.title || '');
   357→  const [slug, setSlug] = React.useState(photozone?.slug || '');
   358→  const [slugTouched, setSlugTouched] = React.useState(!!photozone?.slug);
   359→  const [description, setDescription] = React.useState(photozone?.description || '');
   360→  const [category, setCategory] = React.useState<Category>(
   361→    (photozone?.category as Category) || 'FRAME'
   362→  );
   363→  const [occasions, setOccasions] = React.useState<string[]>(
   364→    (photozone?.occasions as string[]) || []
   365→  );
   366→  const [basePrice, setBasePrice] = React.useState<string>(
   367→    photozone?.basePrice != null ? String(photozone.basePrice) : ''
   368→  );
   369→  const [sizeW, setSizeW] = React.useState<string>(
   370→    photozone?.sizeW != null ? String(photozone.sizeW) : ''
   371→  );
   372→  const [sizeH, setSizeH] = React.useState<string>(
   373→    photozone?.sizeH != null ? String(photozone.sizeH) : ''
   374→  );
   375→  const [imagesText, setImagesText] = React.useState<string>(
   376→    Array.isArray(photozone?.images) ? (photozone!.images as string[]).join('\n') : ''
   377→  );
   378→  const [videosText, setVideosText] = React.useState<string>(
   379→    Array.isArray(photozone?.videos) ? (photozone!.videos as string[]).join('\n') : ''
   380→  );
   381→  const [includesText, setIncludesText] = React.useState<string>(
   382→    Array.isArray(photozone?.includes) ? (photozone!.includes as string[]).join('\n') : ''
   383→  );
   384→  const [badge, setBadge] = React.useState<string>(photozone?.badge || '');
   385→  const [hit, setHit] = React.useState<boolean>(!!photozone?.hit);
   386→  const [sortOrder, setSortOrder] = React.useState<string>(
   387→    String(photozone?.sortOrder ?? 100)
   388→  );
   389→  const [isActive, setIsActive] = React.useState<boolean>(
   390→    photozone?.isActive !== false
   391→  );
   392→  const [elements, setElements] = React.useState<Element[]>(
   393→    (photozone?.elements as any) || []
   394→  );
   395→  const [newElName, setNewElName] = React.useState('');
   396→  const [newElPrice, setNewElPrice] = React.useState('');
   397→  const [newElCategory, setNewElCategory] = React.useState('decor');
   398→  const [newElImage, setNewElImage] = React.useState('');
   399→  const [newElDesc, setNewElDesc] = React.useState('');
   400→  const [saving, setSaving] = React.useState(false);
   401→
   402→  // автоген slug
   403→  React.useEffect(() => {
   404→    if (!slugTouched) {
   405→      setSlug(slugify(title));
   406→    }
   407→  }, [title, slugTouched]);
   408→
   409→  const toggleOccasion = (val: string) => {
   410→    setOccasions((prev) =>
   411→      prev.includes(val) ? prev.filter((o) => o !== val) : [...prev, val]
   412→    );
   413→  };
   414→
   415→  const addElement = () => {
   416→    if (!newElName.trim()) return;
   417→    setElements((prev) => [
   418→      ...prev,
   419→      {
   420→        name: newElName.trim(),
   421→        price: Number(newElPrice) || 0,
   422→        category: newElCategory,
   423→        image: newElImage.trim() || null,
   424→        description: newElDesc.trim() || null,
   425→      } as any,
   426→    ]);
   427→    setNewElName('');
   428→    setNewElPrice('');
   429→    setNewElCategory('decor');
   430→    setNewElImage('');
   431→    setNewElDesc('');
   432→  };
   433→
   434→  const removeElement = (idx: number) => {
   435→    setElements((prev) => prev.filter((_, i) => i !== idx));
   436→  };
   437→
   438→  const save = async () => {
   439→    if (!title || !slug || !category || basePrice === '') {
   440→      toast.error('Заполните название, slug, категорию и цену');
   441→      return;
   442→    }
   443→    const payload: any = {
   444→      title: title.trim(),
   445→      slug: slug.trim(),
   446→      description: description.trim(),
   447→      category,
   448→      occasions,
   449→      basePrice: Number(basePrice),
   450→      sizeW: sizeW ? Number(sizeW) : null,
   451→      sizeH: sizeH ? Number(sizeH) : null,
   452→      images: imagesText
   453→        .split('\n')
   454→        .map((s) => s.trim())
   455→        .filter(Boolean),
   456→      videos: videosText
   457→        .split('\n')
   458→        .map((s) => s.trim())
   459→        .filter(Boolean),
   460→      includes: includesText
   461→        .split('\n')
   462→        .map((s) => s.trim())
   463→        .filter(Boolean),
   464→      badge: badge || null,
   465→      hit,
   466→      sortOrder: Number(sortOrder) || 100,
   467→      isActive,
   468→      elements: elements.map((e) => ({
   469→        id: (e as any).id,
   470→        name: e.name,
   471→        price: Number(e.price) || 0,
   472→        isActive: (e as any).isActive !== false,
   473→        category: (e as any).category || 'decor',
   474→        image: (e as any).image || null,
   475→        description: (e as any).description || null,
   476→      })),
   477→    };
   478→    setSaving(true);
   479→    try {
   480→      if (isEdit && photozone) {
   481→        await api(`/api/admin/photozones/${photozone.id}`, {
   482→          method: 'PATCH',
   483→          body: JSON.stringify(payload),
   484→        });
   485→        toast.success('Фотозона обновлена');
   486→      } else {
   487→        await api('/api/admin/photozones', {
   488→          method: 'POST',
   489→          body: JSON.stringify(payload),
   490→        });
   491→        toast.success('Фотозона создана');
   492→      }
   493→      onSaved();
   494→    } catch (e: any) {
   495→      if (e?.status === 409) toast.error('Slug уже занят');
   496→      else toast.error(e?.message || 'Не удалось сохранить');
   497→    } finally {
   498→      setSaving(false);
   499→    }
   500→  };
   501→
   502→  return (
   503→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   504→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto">
   505→        <DialogHeader>
   506→          <DialogTitle>
   507→            {isEdit ? 'Редактирование фотозоны' : 'Новая фотозона'}
   508→          </DialogTitle>
   509→          <DialogDescription>
   510→            Поля со звёздочкой — обязательные. Slug генерируется автоматически из названия.
   511→          </DialogDescription>
   512→        </DialogHeader>
   513→
   514→        <div className="space-y-4">
   515→          {/* Основное */}
   516→          <div className="grid sm:grid-cols-2 gap-3">
   517→            <div className="space-y-1.5">
   518→              <Label>Название *</Label>
   519→              <Input
   520→                value={title}
   521→                onChange={(e) => setTitle(e.target.value)}
   522→                placeholder="Например, Love-инсталляция"
   523→              />
   524→            </div>
   525→            <div className="space-y-1.5">
   526→              <Label>Slug *</Label>
   527→              <Input
   528→                value={slug}
   529→                onChange={(e) => {
   530→                  setSlug(e.target.value);
   531→                  setSlugTouched(true);
   532→                }}
   533→                placeholder="love-installyaciya"
   534→                className="font-mono text-sm"
   535→              />
   536→            </div>
   537→          </div>
   538→
   539→          <div className="space-y-1.5">
   540→            <Label>Описание</Label>
   541→            <Textarea
   542→              rows={3}
   543→              value={description}
   544→              onChange={(e) => setDescription(e.target.value)}
   545→              placeholder="Короткое описание для карточки и страницы фотозоны"
   546→            />
   547→          </div>
   548→
   549→          <div className="grid sm:grid-cols-3 gap-3">
   550→            <div className="space-y-1.5">
   551→              <Label>Категория *</Label>
   552→              <Select value={category} onValueChange={(v) => setCategory(v as Category)}>
   553→                <SelectTrigger className="w-full">
   554→                  <SelectValue />
   555→                </SelectTrigger>
   556→                <SelectContent>
   557→                  {CATEGORIES.map((c) => (
   558→                    <SelectItem key={c.value} value={c.value}>
   559→                      {c.label}
   560→                    </SelectItem>
   561→                  ))}
   562→                </SelectContent>
   563→              </Select>
   564→            </div>
   565→            <div className="space-y-1.5">
   566→              <Label>Цена (от), ₽ *</Label>
   567→              <Input
   568→                type="number"
   569→                value={basePrice}
   570→                onChange={(e) => setBasePrice(e.target.value)}
   571→                placeholder="12000"
   572→              />
   573→            </div>
   574→            <div className="space-y-1.5">
   575→              <Label>Бейдж</Label>
   576→              <Select value={badge} onValueChange={setBadge}>
   577→                <SelectTrigger className="w-full">
   578→                  <SelectValue />
