    74→function slugify(s: string) {
    75→  return (s || '')
    76→    .toLowerCase()
    77→    .replace(/[^a-z0-9а-я]+/gi, '-')
    78→    .replace(/^-+|-+$/g, '')
    79→    .slice(0, 80);
    80→}

   343→// ============ Редактор фотозоны ============

   345→
   346→function PhotozoneEditor({
   347→  photozone,
   348→  onClose,
   349→  onSaved,
   350→}: {
   351→  photozone: PhotoZone | null;
   352→  onClose: () => void;
   353→  onSaved: () => void;
   354→}) {
   355→  const isEdit = !!photozone;
   356→  const [title, setTitle] = React.useState(photozone?.title || '');
   357→  const [slug, setSlug] = React.useState(photozone?.slug || '');
   358→  const [slugTouched, setSlugTouched] = React.useState(!!photozone?.slug);
   359→  const [description, setDescription] = React.useState(photozone?.description || '');
   360→  const [category, setCategory] = React.useState<Category>(
   361→    (photozone?.category as Category) || 'FRAME'
   362→  );
   363→  const [occasions, setOccasions] = React.useState<string[]>(
   364→    (photozone?.occasions as string[]) || []
   365→  );
   366→  const [basePrice, setBasePrice] = React.useState<string>(
   367→    photozone?.basePrice != null ? String(photozone.basePrice) : ''
   368→  );
   369→  const [sizeW, setSizeW] = React.useState<string>(
   370→    photozone?.sizeW != null ? String(photozone.sizeW) : ''
   371→  );
   372→  const [sizeH, setSizeH] = React.useState<string>(
   373→    photozone?.sizeH != null ? String(photozone.sizeH) : ''
   374→  );
   375→  const [imagesText, setImagesText] = React.useState<string>(
   376→    Array.isArray(photozone?.images) ? (photozone!.images as string[]).join('\n') : ''
   377→  );
   378→  const [includesText, setIncludesText] = React.useState<string>(
   379→    Array.isArray(photozone?.includes) ? (photozone!.includes as string[]).join('\n') : ''
   380→  );
   381→  const [badge, setBadge] = React.useState<string>(photozone?.badge || '');
   382→  const [hit, setHit] = React.useState<boolean>(!!photozone?.hit);
   383→  const [sortOrder, setSortOrder] = React.useState<string>(
   384→    String(photozone?.sortOrder ?? 100)
   385→  );
   386→  const [isActive, setIsActive] = React.useState<boolean>(
   387→    photozone?.isActive !== false
   388→  );
   389→  const [elements, setElements] = React.useState<Element[]>(
   390→    (photozone?.elements as any) || []
   391→  );
   392→  const [newElName, setNewElName] = React.useState('');
   393→  const [newElPrice, setNewElPrice] = React.useState('');
   394→  const [saving, setSaving] = React.useState(false);
   395→
   396→  // автоген slug
   397→  React.useEffect(() => {
   398→    if (!slugTouched) {
   399→      setSlug(slugify(title));
   400→    }
   401→  }, [title, slugTouched]);
   402→
   403→  const toggleOccasion = (val: string) => {
   404→    setOccasions((prev) =>
   405→      prev.includes(val) ? prev.filter((o) => o !== val) : [...prev, val]
   406→    );
   407→  };
   408→
   409→  const addElement = () => {
   410→    if (!newElName.trim()) return;
   411→    setElements((prev) => [
   412→      ...prev,
   413→      { name: newElName.trim(), price: Number(newElPrice) || 0 },
   414→    ]);
   415→    setNewElName('');
   416→    setNewElPrice('');
   417→  };
   418→
   419→  const removeElement = (idx: number) => {
   420→    setElements((prev) => prev.filter((_, i) => i !== idx));
   421→  };
   422→
   423→  const save = async () => {
   424→    if (!title || !slug || !category || basePrice === '') {
   425→      toast.error('Заполните название, slug, категорию и цену');
   426→      return;
   427→    }
   428→    const payload: any = {
   429→      title: title.trim(),
   430→      slug: slug.trim(),
   431→      description: description.trim(),
   432→      category,
   433→      occasions,
   434→      basePrice: Number(basePrice),
   435→      sizeW: sizeW ? Number(sizeW) : null,
   436→      sizeH: sizeH ? Number(sizeH) : null,
   437→      images: imagesText
   438→        .split('\n')
   439→        .map((s) => s.trim())
   440→        .filter(Boolean),
   441→      includes: includesText
   442→        .split('\n')
   443→        .map((s) => s.trim())
   444→        .filter(Boolean),
   445→      badge: badge || null,
   446→      hit,
   447→      sortOrder: Number(sortOrder) || 100,
   448→      isActive,
   449→    };
   450→    setSaving(true);
   451→    try {
   452→      if (isEdit && photozone) {
   453→        await api(`/api/admin/photozones/${photozone.id}`, {
   454→          method: 'PATCH',
   455→          body: JSON.stringify(payload),
   456→        });
   457→        toast.success('Фотозона обновлена');
   458→      } else {
   459→        await api('/api/admin/photozones', {
   460→          method: 'POST',
   461→          body: JSON.stringify(payload),
   462→        });
   463→        toast.success('Фотозона создана');
   464→      }
   465→      onSaved();
   466→    } catch (e: any) {
   467→      if (e?.status === 409) toast.error('Slug уже занят');
   468→      else toast.error(e?.message || 'Не удалось сохранить');
   469→    } finally {
   470→      setSaving(false);
   471→    }
   472→  };
   473→
   474→  return (
   475→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   476→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto">
   477→        <DialogHeader>
   478→          <DialogTitle>
   479→            {isEdit ? 'Редактирование фотозоны' : 'Новая фотозона'}
   480→          </DialogTitle>
   481→          <DialogDescription>
   482→            Поля со звёздочкой — обязательные. Slug генерируется автоматически из названия.
   483→          </DialogDescription>
   484→        </DialogHeader>
   485→
   486→        <div className="space-y-4">
   487→          {/* Основное */}
   488→          <div className="grid sm:grid-cols-2 gap-3">
   489→            <div className="space-y-1.5">
   490→              <Label>Название *</Label>
   491→              <Input
   492→                value={title}
   493→                onChange={(e) => setTitle(e.target.value)}
   494→                placeholder="Например, Love-инсталляция"
   495→              />
   496→            </div>
   497→            <div className="space-y-1.5">
   498→              <Label>Slug *</Label>
   499→              <Input
   500→                value={slug}
   501→                onChange={(e) => {
   502→                  setSlug(e.target.value);
   503→                  setSlugTouched(true);
   504→                }}
   505→                placeholder="love-installyaciya"
   506→                className="font-mono text-sm"
   507→              />
   508→            </div>
   509→          </div>
   510→
   511→          <div className="space-y-1.5">
   512→            <Label>Описание</Label>
   513→            <Textarea
   514→              rows={3}
   515→              value={description}
   516→              onChange={(e) => setDescription(e.target.value)}
   517→              placeholder="Короткое описание для карточки и страницы фотозоны"
   518→            />
   519→          </div>
   520→
   521→          <div className="grid sm:grid-cols-3 gap-3">
   522→            <div className="space-y-1.5">
   523→              <Label>Категория *</Label>
   524→              <Select value={category} onValueChange={(v) => setCategory(v as Category)}>
   525→                <SelectTrigger className="w-full">
   526→                  <SelectValue />
   527→                </SelectTrigger>
   528→                <SelectContent>
   529→                  {CATEGORIES.map((c) => (
   530→                    <SelectItem key={c.value} value={c.value}>
   531→                      {c.label}
   532→                    </SelectItem>
   533→                  ))}
   534→                </SelectContent>
   535→              </Select>
   536→            </div>
   537→            <div className="space-y-1.5">
   538→              <Label>Цена (от), ₽ *</Label>
   539→              <Input
   540→                type="number"
   541→                value={basePrice}
   542→                onChange={(e) => setBasePrice(e.target.value)}
   543→                placeholder="12000"
   544→              />
   545→            </div>
   546→            <div className="space-y-1.5">
   547→              <Label>Бейдж</Label>
   548→              <Select value={badge} onValueChange={setBadge}>
   549→                <SelectTrigger className="w-full">
   550→                  <SelectValue />
   551→                </SelectTrigger>
   552→                <SelectContent>
   553→                  {BADGES.map((b) => (
   554→                    <SelectItem key={b.value || 'none'} value={b.value || 'none'}>
   555→                      {b.label}
   556→                    </SelectItem>
   557→                  ))}
   558→                </SelectContent>
   559→              </Select>
   560→            </div>
   561→          </div>
   562→
   563→          <div className="grid sm:grid-cols-3 gap-3">
   564→            <div className="space-y-1.5">
   565→              <Label>Ширина, м</Label>
   566→              <Input
   567→                type="number"
   568→                step="0.1"
   569→                value={sizeW}
   570→                onChange={(e) => setSizeW(e.target.value)}
   571→                placeholder="3"
   572→              />
   573→            </div>
   574→            <div className="space-y-1.5">
   575→              <Label>Высота, м</Label>
   576→              <Input
   577→                type="number"
   578→                step="0.1"
   579→                value={sizeH}
   580→                onChange={(e) => setSizeH(e.target.value)}
   581→                placeholder="2.2"
   582→              />
