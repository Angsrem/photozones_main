    69→    69→
    70→    70→export function CalculatorWizard() {
    71→    71→  const [step, setStep] = useState(0); // 0..3
    72→    72→  const [calculated, setCalculated] = useState(false);
    73→    73→
    74→    74→  const [occasion, setOccasion] = useState<string>('');
    75→    75→  const [style, setStyle] = useState<string>('');
    76→    76→  const [base, setBase] = useState<string>('');
    77→    77→  const [addons, setAddons] = useState<string[]>([]);
    78→    78→  const [date, setDate] = useState<string>('');
    79→    79→  const [address, setAddress] = useState<string>('');
    80→    80→  const [isSuburb, setIsSuburb] = useState(false);
    81→    81→
    82→    82→  // min date — считаем в рендере; добавляем suppressHydrationWarning на input,
    83→    83→  // чтобы избежать предупреждения о разнице SSR/CSR на границе суток.
    84→    84→  const minDate = todayISO();
    85→    85→
    86→    86→  const occasionOpt = OCCASION_OPTIONS.find((o) => o.value === occasion);
    87→    87→  const styleOpt = STYLE_OPTIONS.find((s) => s.value === style);
    88→    88→  const baseOpt = BASE_OPTIONS.find((b) => b.value === base);
    89→    89→  const selectedAddons = ADDON_OPTIONS.filter((a) => addons.includes(a.value));
    90→    90→
    91→    91→  const breakdown = useMemo(() => {
    92→    92→    const basePrice = baseOpt?.price ?? 0;
    93→    93→    const styleModifier = styleOpt?.modifier ?? 0;
    94→    94→    const addonsTotal = selectedAddons.reduce((s, a) => s + a.price, 0);
    95→    95→    const suburb = isSuburb ? SITE.suburbSurcharge : 0;
    96→    96→    const total = basePrice + styleModifier + addonsTotal + suburb;
    97→    97→    return { basePrice, styleModifier, addonsTotal, suburb, total };
    98→    98→  }, [baseOpt, styleOpt, selectedAddons, isSuburb]);
    99→    99→
   100→   100→  const canNext = () => {
   101→   101→    if (step === 0) return !!occasion;
   102→   102→    if (step === 1) return !!style;
   103→   103→    if (step === 2) return !!base;
   104→   104→    if (step === 3) return !!date;
   105→   105→    return false;
   106→   106→  };
   107→   107→
   108→   108→  const next = () => {
   109→   109→    if (step < 3) setStep(step + 1);
   110→   110→    else setCalculated(true);
   111→   111→  };
   112→   112→  const back = () => {
   113→   113→    if (calculated) { setCalculated(false); return; }
   114→   114→    if (step > 0) setStep(step - 1);
   115→   115→  };
   116→   116→
   117→   117→  const toggleAddon = (v: string) => {
   118→   118→    setAddons((prev) => (prev.includes(v) ? prev.filter((x) => x !== v) : [...prev, v]));
   119→   119→  };
   120→   120→
   121→   121→  const handleOrder = () => {
   122→   122→    const config = {
   123→   123→      occasion: occasionOpt?.value ?? occasion,
   124→   124→      occasionLabel: occasionOpt?.label ?? occasion,
   125→   125→      style: styleOpt?.value ?? style,
   126→   126→      styleLabel: styleOpt?.label ?? style,
   127→   127→      base: baseOpt
   128→   128→        ? { value: baseOpt.value, label: baseOpt.label, desc: baseOpt.desc, price: baseOpt.price }
   129→   129→        : null,
   130→   130→      addons: selectedAddons.map((a) => ({ value: a.value, label: a.label, price: a.price })),
   131→   131→      date,
   132→   132→      address,
   133→   133→      isSuburb,
   134→   134→      suburbSurcharge: isSuburb ? SITE.suburbSurcharge : 0,
   135→   135→      breakdown: {
   136→   136→        base: breakdown.basePrice,
   137→   137→        style: breakdown.styleModifier,
   138→   138→        addons: breakdown.addonsTotal,
   139→   139→        suburb: breakdown.suburb,
   140→   140→        total: breakdown.total,
   141→   141→      },
   142→   142→    };
   143→   143→    openQuickOrder({
   144→   144→      source: 'kalkulyator',
   145→   145→      sourcePage: '/kalkulyator',
   146→   146→      occasion: occasionOpt?.value ?? occasion,
   147→   147→      priceEstimate: breakdown.total,
   148→   148→      config,
   149→   149→    });
   150→   150→  };
   151→   151→
   152→   152→  const progressValue = calculated ? 100 : ((step + 1) / 4) * 100;
   153→   153→
   154→   154→  return (
   155→   155→    <div className="rounded-3xl border border-border bg-card/60 backdrop-blur p-4 sm:p-6 lg:p-8 shadow-2xl">
   156→   156→      {/* ===== Stepper / Progress ===== */}
   157→   157→      <div className="mb-6">
   158→   158→        <div className="flex items-center justify-between gap-1 mb-3">
   159→   159→          {STEPS.map((label, i) => {
   160→   160→            const done = calculated || i < step;
   161→   161→            const active = !calculated && i === step;
   162→   162→            return (
   163→   163→              <div key={label} className="flex-1 flex items-center gap-1 min-w-0">
   164→   164→                <div
   165→   165→                  className={[
   166→   166→                    'flex items-center gap-2 rounded-full px-2 sm:px-3 h-9 sm:h-10 text-xs sm:text-sm transition-colors whitespace-nowrap',
   167→   167→                    active ? 'bg-gold text-primary-foreground font-semibold'
   168→   168→                      : done ? 'bg-gold/15 text-gold'
   169→   169→                      : 'bg-muted/50 text-muted-foreground',
   170→   170→                  ].join(' ')}
   171→   171→                >
   172→   172→                  <span className="flex items-center justify-center w-5 h-5 rounded-full text-[11px]">
   173→   173→                    {done ? <Check className="w-3.5 h-3.5" /> : i + 1}
   174→   174→                  </span>
   175→   175→                  <span className="hidden sm:inline">{label}</span>
   176→   176→                </div>
   177→   177→                {i < STEPS.length - 1 && (
   178→   178→                  <div className="flex-1 h-px bg-border hidden sm:block" />
   179→   179→                )}
   180→   180→              </div>
   181→   181→            );
   182→   182→          })}
   183→   183→        </div>
   184→   184→        <Progress value={progressValue} className="h-1.5 bg-muted/50 [&>div]:bg-gold" />
   185→   185→      </div>
   186→   186→
   187→   187→      {/* ===== Body ===== */}
   188→   188→      <div className="min-h-[340px]">
   189→   189→        <AnimatePresence mode="wait">
   190→   190→          {calculated ? (
   191→   191→            <motion.div
   192→   192→              key="result"
   193→   193→              initial={{ opacity: 0, y: 16 }}
   194→   194→              animate={{ opacity: 1, y: 0 }}
   195→   195→              exit={{ opacity: 0, y: -16 }}
   196→   196→              transition={{ duration: 0.3 }}
   197→   197→            >
   198→   198→              <ResultView
   199→   199→                occasionLabel={occasionOpt?.label ?? occasion}
   200→   200→                styleLabel={styleOpt?.label ?? style}
   201→   201→                baseOpt={baseOpt}
   202→   202→                selectedAddons={selectedAddons}
   203→   203→                date={date}
   204→   204→                address={address}
   205→   205→                isSuburb={isSuburb}
   206→   206→                breakdown={breakdown}
   207→   207→                onOrder={handleOrder}
   208→   208→                onEdit={() => setCalculated(false)}
   209→   209→              />
   210→   210→            </motion.div>
   211→   211→          ) : (
   212→   212→            <motion.div
   213→   213→              key={step}
   214→   214→              initial={{ opacity: 0, x: 24 }}
   215→   215→              animate={{ opacity: 1, x: 0 }}
   216→   216→              exit={{ opacity: 0, x: -24 }}
   217→   217→              transition={{ duration: 0.25 }}
   218→   218→            >
   219→   219→              {step === 0 && (
   220→   220→                <StepOccasion value={occasion} onChange={setOccasion} />
   221→   221→              )}
   222→   222→              {step === 1 && (
   223→   223→                <StepStyle value={style} onChange={setStyle} />
   224→   224→              )}
   225→   225→              {step === 2 && (
   226→   226→                <StepBaseAndAddons
   227→   227→                  base={base} onBase={setBase}
   228→   228→                  addons={addons} onToggleAddon={toggleAddon}
   229→   229→                />
   230→   230→              )}
   231→   231→              {step === 3 && (
   232→   232→                <StepDateAddress
   233→   233→                  date={date} onDate={setDate}
   234→   234→                  address={address} onAddress={setAddress}
   235→   235→                  isSuburb={isSuburb} onSuburb={setIsSuburb}
   236→   236→                  minDate={minDate}
   237→   237→                />
   238→   238→              )}
   239→   239→            </motion.div>
   240→   240→          )}
   241→   241→        </AnimatePresence>
   242→   242→      </div>
   243→   243→
   244→   244→      {/* ===== Nav buttons ===== */}
   245→   245→      <div className="flex items-center justify-between gap-3 mt-6 pt-5 border-t border-border/60">
   246→   246→        <Button
   247→   247→          variant="ghost"
   248→   248→          onClick={back}
   249→   249→          disabled={!calculated && step === 0}
   250→   250→          className="h-12 px-5"
   251→   251→        >
   252→   252→          <ArrowLeft className="w-4 h-4 mr-1" /> Назад
   253→   253→        </Button>
   254→   254→
   255→   255→        {!calculated ? (
   256→   256→          <Button
   257→   257→            onClick={next}
   258→   258→            disabled={!canNext()}
   259→   259→            className="h-12 px-7 bg-gold text-primary-foreground hover:bg-gold/90 font-semibold"
   260→   260→          >
   261→   261→            {step === 3 ? <>Рассчитать <Sparkles className="w-4 h-4 ml-1" /></> : <>Далее <ArrowRight className="w-4 h-4 ml-1" /></>}
   262→   262→          </Button>
   263→   263→        ) : (
   264→   264→          <Button
   265→   265→            onClick={handleOrder}
   266→   266→            className="h-12 px-7 bg-gold text-primary-foreground hover:bg-gold/90 font-semibold"
   267→   267→          >
   268→   268→            <Sparkles className="w-4 h-4 mr-1" /> Зафиксировать расчёт и заказать
   269→   269→          </Button>
   270→   270→        )}
   271→   271→      </div>
   272→   272→    </div>
   273→   273→  );
   274→   274→}
