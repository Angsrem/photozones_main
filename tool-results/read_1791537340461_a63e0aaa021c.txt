    68→/* ============ Компонент ============ */
    69→
    70→export function CalculatorWizard() {
    71→  const [step, setStep] = useState(0); // 0..3
    72→  const [calculated, setCalculated] = useState(false);
    73→
    74→  const [occasion, setOccasion] = useState<string>('');
    75→  const [style, setStyle] = useState<string>('');
    76→  const [base, setBase] = useState<string>('');
    77→  const [addons, setAddons] = useState<string[]>([]);
    78→  const [date, setDate] = useState<string>('');
    79→  const [address, setAddress] = useState<string>('');
    80→  const [isSuburb, setIsSuburb] = useState(false);
    81→
    82→  // min date — считаем в рендере; добавляем suppressHydrationWarning на input,
    83→  // чтобы избежать предупреждения о разнице SSR/CSR на границе суток.
    84→  const minDate = todayISO();
    85→
    86→  const occasionOpt = OCCASION_OPTIONS.find((o) => o.value === occasion);
    87→  const styleOpt = STYLE_OPTIONS.find((s) => s.value === style);
    88→  const baseOpt = BASE_OPTIONS.find((b) => b.value === base);
    89→  const selectedAddons = ADDON_OPTIONS.filter((a) => addons.includes(a.value));
    90→
    91→  const breakdown = useMemo(() => {
    92→    const basePrice = baseOpt?.price ?? 0;
    93→    const styleModifier = styleOpt?.modifier ?? 0;
    94→    const addonsTotal = selectedAddons.reduce((s, a) => s + a.price, 0);
    95→    const suburb = isSuburb ? SITE.suburbSurcharge : 0;
    96→    const total = basePrice + styleModifier + addonsTotal + suburb;
    97→    return { basePrice, styleModifier, addonsTotal, suburb, total };
    98→  }, [baseOpt, styleOpt, selectedAddons, isSuburb]);
    99→
   100→  const canNext = () => {
   101→    if (step === 0) return !!occasion;
   102→    if (step === 1) return !!style;
   103→    if (step === 2) return !!base;
   104→    if (step === 3) return !!date;
   105→    return false;
   106→  };
   107→
   108→  const next = () => {
   109→    if (step < 3) setStep(step + 1);
   110→    else setCalculated(true);
   111→  };
   112→  const back = () => {
   113→    if (calculated) { setCalculated(false); return; }
   114→    if (step > 0) setStep(step - 1);
   115→  };
   116→
   117→  const toggleAddon = (v: string) => {
   118→    setAddons((prev) => (prev.includes(v) ? prev.filter((x) => x !== v) : [...prev, v]));
   119→  };
   120→
   121→  const handleOrder = () => {
   122→    const config = {
   123→      occasion: occasionOpt?.value ?? occasion,
   124→      occasionLabel: occasionOpt?.label ?? occasion,
   125→      style: styleOpt?.value ?? style,
   126→      styleLabel: styleOpt?.label ?? style,
   127→      base: baseOpt
   128→        ? { value: baseOpt.value, label: baseOpt.label, desc: baseOpt.desc, price: baseOpt.price }
   129→        : null,
   130→      addons: selectedAddons.map((a) => ({ value: a.value, label: a.label, price: a.price })),
   131→      date,
   132→      address,
   133→      isSuburb,
   134→      suburbSurcharge: isSuburb ? SITE.suburbSurcharge : 0,
   135→      breakdown: {
   136→        base: breakdown.basePrice,
   137→        style: breakdown.styleModifier,
   138→        addons: breakdown.addonsTotal,
   139→        suburb: breakdown.suburb,
   140→        total: breakdown.total,
   141→      },
   142→    };
   143→    openQuickOrder({
   144→      source: 'kalkulyator',
   145→      sourcePage: '/kalkulyator',
   146→      occasion: occasionOpt?.value ?? occasion,
   147→      priceEstimate: breakdown.total,
   148→      config,
   149→    });
   150→  };
   151→
   152→  const progressValue = calculated ? 100 : ((step + 1) / 4) * 100;
   153→
   154→  return (
   155→    <div className="rounded-3xl border border-border bg-card/60 backdrop-blur p-4 sm:p-6 lg:p-8 shadow-2xl">
   156→      {/* ===== Stepper / Progress ===== */}
   157→      <div className="mb-6">
   158→        <div className="flex items-center justify-between gap-1 mb-3">
   159→          {STEPS.map((label, i) => {
   160→            const done = calculated || i < step;
   161→            const active = !calculated && i === step;
   162→            return (
   163→              <div key={label} className="flex-1 flex items-center gap-1 min-w-0">
   164→                <div
   165→                  className={[
   166→                    'flex items-center gap-2 rounded-full px-2 sm:px-3 h-9 sm:h-10 text-xs sm:text-sm transition-colors whitespace-nowrap',
   167→                    active ? 'bg-gold text-primary-foreground font-semibold'
   168→                      : done ? 'bg-gold/15 text-gold'
   169→                      : 'bg-muted/50 text-muted-foreground',
   170→                  ].join(' ')}
   171→                >
   172→                  <span className="flex items-center justify-center w-5 h-5 rounded-full text-[11px]">
   173→                    {done ? <Check className="w-3.5 h-3.5" /> : i + 1}
   174→                  </span>
   175→                  <span className="hidden sm:inline">{label}</span>
   176→                </div>
   177→                {i < STEPS.length - 1 && (
   178→                  <div className="flex-1 h-px bg-border hidden sm:block" />
   179→                )}
   180→              </div>
   181→            );
   182→          })}
   183→        </div>
   184→        <Progress value={progressValue} className="h-1.5 bg-muted/50 [&>div]:bg-gold" />
   185→      </div>
   186→
   187→      {/* ===== Body ===== */}
   188→      <div className="min-h-[340px]">
   189→        <AnimatePresence mode="wait">
   190→          {calculated ? (
   191→            <motion.div
   192→              key="result"
   193→              initial={{ opacity: 0, y: 16 }}
   194→              animate={{ opacity: 1, y: 0 }}
   195→              exit={{ opacity: 0, y: -16 }}
   196→              transition={{ duration: 0.3 }}
   197→            >
   198→              <ResultView
   199→                occasionLabel={occasionOpt?.label ?? occasion}
   200→                styleLabel={styleOpt?.label ?? style}
   201→                baseOpt={baseOpt}
   202→                selectedAddons={selectedAddons}
   203→                date={date}
   204→                address={address}
   205→                isSuburb={isSuburb}
   206→                breakdown={breakdown}
   207→                onOrder={handleOrder}
   208→                onEdit={() => setCalculated(false)}
   209→              />
   210→            </motion.div>
   211→          ) : (
   212→            <motion.div
   213→              key={step}
   214→              initial={{ opacity: 0, x: 24 }}
   215→              animate={{ opacity: 1, x: 0 }}
   216→              exit={{ opacity: 0, x: -24 }}
   217→              transition={{ duration: 0.25 }}
   218→            >
   219→              {step === 0 && (
   220→                <StepOccasion value={occasion} onChange={setOccasion} />
   221→              )}
   222→              {step === 1 && (
   223→                <StepStyle value={style} onChange={setStyle} />
   224→              )}
   225→              {step === 2 && (
   226→                <StepBaseAndAddons
   227→                  base={base} onBase={setBase}
   228→                  addons={addons} onToggleAddon={toggleAddon}
   229→                />
   230→              )}
   231→              {step === 3 && (
   232→                <StepDateAddress
   233→                  date={date} onDate={setDate}
   234→                  address={address} onAddress={setAddress}
   235→                  isSuburb={isSuburb} onSuburb={setIsSuburb}
   236→                  minDate={minDate}
   237→                />
   238→              )}
   239→            </motion.div>
   240→          )}
   241→        </AnimatePresence>
   242→      </div>
   243→
   244→      {/* ===== Nav buttons ===== */}
   245→      <div className="flex items-center justify-between gap-3 mt-6 pt-5 border-t border-border/60">
   246→        <Button
   247→          variant="ghost"
   248→          onClick={back}
   249→          disabled={!calculated && step === 0}
   250→          className="h-12 px-5"
   251→        >
   252→          <ArrowLeft className="w-4 h-4 mr-1" /> Назад
   253→        </Button>
   254→
   255→        {!calculated ? (
   256→          <Button
   257→            onClick={next}
   258→            disabled={!canNext()}
   259→            className="h-12 px-7 bg-gold text-primary-foreground hover:bg-gold/90 font-semibold"
   260→          >
   261→            {step === 3 ? <>Рассчитать <Sparkles className="w-4 h-4 ml-1" /></> : <>Далее <ArrowRight className="w-4 h-4 ml-1" /></>}
   262→          </Button>
   263→        ) : (
   264→          <Button
   265→            onClick={handleOrder}
   266→            className="h-12 px-7 bg-gold text-primary-foreground hover:bg-gold/90 font-semibold"
   267→          >
   268→            <Sparkles className="w-4 h-4 mr-1" /> Зафиксировать расчёт и заказать
   269→          </Button>
   270→        )}
   271→      </div>
   272→    </div>
   273→  );
   274→}
