    72→// строит сетку месяца (с доп. ячейками из соседних месяцев чтобы заполнить 6 строк × 7 столбцов)
    73→function buildMonthGrid(year: number, month: number) {
    74→  // month: 0-based
    75→  const first = new Date(year, month, 1);
    76→  const firstDayJs = new Date(year, month, 1);
    77→  // делаем понедельник первым днём недели
    78→  const firstWeekday = (firstDayJs.getDay() + 6) % 7; // 0=Пн
    79→  const daysInMonth = new Date(year, month + 1, 0).getDate();
    80→  const cells: { date: string; day: number; inMonth: boolean; isToday: boolean }[] = [];
    81→  const todayStr = new Date().toISOString().slice(0, 10);
    82→
    83→  // префикс из прошлого месяца
    84→  for (let i = firstWeekday - 1; i >= 0; i--) {
    85→    const d = new Date(year, month, -i);
    86→    cells.push({
    87→      date: d.toISOString().slice(0, 10),
    88→      day: d.getDate(),
    89→      inMonth: false,
    90→      isToday: d.toISOString().slice(0, 10) === todayStr,
    91→    });
    92→  }
    93→  for (let i = 1; i <= daysInMonth; i++) {
    94→    const d = new Date(year, month, i);
    95→    const ds = d.toISOString().slice(0, 10);
    96→    cells.push({ date: ds, day: i, inMonth: true, isToday: ds === todayStr });
    97→  }
    98→  // суффикс до 42 ячеек (6 строк)
    99→  let i = 1;
   100→  while (cells.length < 42) {
   101→    const d = new Date(year, month + 1, i++);
   102→    cells.push({
   103→      date: d.toISOString().slice(0, 10),
   104→      day: d.getDate(),
   105→      inMonth: false,
   106→      isToday: d.toISOString().slice(0, 10) === todayStr,
   107→    });
   108→  }
   109→  return cells;
   110→}

   539→
   540→              {bookings.length === 0 ? (
   541→                <div className="text-xs text-muted-foreground py-2">Нет бронирований</div>
   542→              ) : (
   543→                <div className="space-y-1.5">
   544→                  {bookings.map((b) => (
   545→                    <div
   546→                      key={b.id}
   547→                      className="flex items-center justify-between gap-3 rounded-md border p-2.5 text-sm"

   549→                      <div className="flex items-center gap-2 min-w-0">

   775→          </Button>

   778→    </Dialog>

   782→// ============ Форма блокировки ============
   783→function BlockForm({
   784→  date,
   785→  photozones,
   786→  onDone,
   787→}: {
   788→  date: string;
   789→  photozones: PhotoZone[];
   790→  onDone: () => void;
   791→}) {
   792→  const [photozoneId, setPhotozoneId] = React.useState<string>('');
   793→  const [reason, setReason] = React.useState('service');
   794→  const [saving, setSaving] = React.useState(false);
   795→
   796→  const submit = async () => {
   797→    setSaving(true);
   798→    try {
   799→      await api('/api/admin/bookings', {
   800→        method: 'POST',
   801→        body: JSON.stringify({
   802→          type: 'block',
   803→          date,
   804→          photozoneId: photozoneId || null,
   805→          reason,
   806→        }),
   807→      });
   808→      toast.success('Дата заблокирована');
   809→      onDone();
   810→    } catch (e: any) {
   811→      if (e?.status === 409) toast.error('Уже заблокировано');
   812→      else toast.error(e?.message || 'Ошибка');
   813→    } finally {
   814→      setSaving(false);
   815→    }
   816→  };
   817→
   818→  return (
   819→    <div className="rounded-md border border-rose-200 bg-rose-50/50 p-3 space-y-2 mb-3">
   820→      <div className="grid sm:grid-cols-2 gap-2">
   821→        <div className="space-y-1">
   822→          <Label className="text-xs">Фотозона</Label>
   823→          <Select value={photozoneId || '__all__'} onValueChange={(v) => setPhotozoneId(v === '__all__' ? '' : v)}>
   824→            <SelectTrigger className="h-8 text-xs">
   825→              <SelectValue placeholder="Все фотозоны" />
   826→            </SelectTrigger>
   827→            <SelectContent>
   828→              <SelectItem value="__all__">— Все фотозоны —</SelectItem>
   829→              {photozones.map((p) => (
   830→                <SelectItem key={p.id} value={p.id}>
   831→                  {p.title}
   832→                </SelectItem>
   833→              ))}
   834→            </SelectContent>
   835→          </Select>
   836→        </div>
   837→        <div className="space-y-1">
   838→          <Label className="text-xs">Причина</Label>
   839→          <Select value={reason} onValueChange={setReason}>
   840→            <SelectTrigger className="h-8 text-xs">
   841→              <SelectValue />
   842→            </SelectTrigger>
   843→            <SelectContent>
   844→              <SelectItem value="service">Техобслуживание</SelectItem>
   845→              <SelectItem value="reserved">Выкуп</SelectItem>
   846→              <SelectItem value="other">Другое</SelectItem>
   847→            </SelectContent>
   848→          </Select>
   849→        </div>
   850→      </div>
   851→      <div className="flex justify-end gap-2">
   852→        <Button size="sm" onClick={submit} disabled={saving} className="gap-1">
   853→          {saving ? <Loader2 className="size-3.5 animate-spin" /> : <Ban className="size-3.5" />}
   854→          Заблокировать
   855→        </Button>
   856→      </div>
   857→    </div>
   858→  );
   859→}

   862→function AddBookingForm({
   863→  date,
   864→  photozones,
   865→  occupiedPzIds,
   866→  leads,
   867→  onDone,
   868→}: {
   869→  date: string;
   870→  photozones: PhotoZone[];
   871→  occupiedPzIds: Set<string>;
   872→  leads: Lead[];
   873→  onDone: () => void;
   874→}) {
   875→  const [photozoneId, setPhotozoneId] = React.useState('');
   876→  const [leadId, setLeadId] = React.useState('');
   877→  const [note, setNote] = React.useState('');
   878→  const [saving, setSaving] = React.useState(false);
   879→
   880→  const submit = async () => {
   881→    if (!photozoneId) {
   882→      toast.error('Выберите фотозону');
   883→      return;
   884→    }
   885→    setSaving(true);
   886→    try {
   887→      await api('/api/admin/bookings', {
   888→        method: 'POST',
   889→        body: JSON.stringify({
   890→          type: 'booking',
   891→          date,
   892→          photozoneId,
   893→          leadId: leadId || null,
   894→          note,
   895→        }),
   896→      });
   897→      toast.success('Бронирование создано');
   898→      onDone();
   899→    } catch (e: any) {
   900→      if (e?.status === 409)
   901→        toast.error('Конфликт: фотозона занята или заблокирована на эту дату');
   902→      else toast.error(e?.message || 'Ошибка');
   903→    } finally {
   904→      setSaving(false);
   905→    }
   906→  };
   907→
   908→  return (
   909→    <div className="rounded-md border p-3 space-y-2 mb-3">
   910→      <div className="grid sm:grid-cols-2 gap-2">
   911→        <div className="space-y-1">
   912→          <Label className="text-xs">Фотозона *</Label>
   913→          <Select value={photozoneId} onValueChange={setPhotozoneId}>
   914→            <SelectTrigger className="h-8 text-xs">
   915→              <SelectValue placeholder="Выбрать" />
   916→            </SelectTrigger>
   917→            <SelectContent>
   918→              {photozones.map((p) => (
   919→                <SelectItem key={p.id} value={p.id} disabled={occupiedPzIds.has(p.id)}>
   920→                  {p.title} {occupiedPzIds.has(p.id) ? '(занята)' : ''}
   921→                </SelectItem>
   922→              ))}
   923→            </SelectContent>
   924→          </Select>
   925→        </div>
   926→        <div className="space-y-1">
   927→          <Label className="text-xs">Заявка (необязательно)</Label>
   928→          <Select value={leadId || '__none__'} onValueChange={(v) => setLeadId(v === '__none__' ? '' : v)}>
   929→            <SelectTrigger className="h-8 text-xs">
   930→              <SelectValue placeholder="Без заявки" />
   931→            </SelectTrigger>
   932→            <SelectContent>
   933→              <SelectItem value="__none__">— Без заявки —</SelectItem>
   934→              {leads.map((l) => (
   935→                <SelectItem key={l.id} value={l.id}>
   936→                  {l.name} · {l.phone}
   937→                </SelectItem>
   938→              ))}
   939→            </SelectContent>
   940→          </Select>
   941→        </div>
   942→      </div>
   943→      <div className="space-y-1">
   944→        <Label className="text-xs">Заметка</Label>
   945→        <Input
   946→          value={note}
   947→          onChange={(e) => setNote(e.target.value)}
   948→          placeholder="Например: договорённость по телефону"
   949→          className="h-8 text-xs"
   950→        />
   951→      </div>
   952→      <div className="flex justify-end gap-2">
   953→        <Button size="sm" onClick={submit} disabled={saving} className="gap-1">
   954→          {saving ? <Loader2 className="size-3.5 animate-spin" /> : <Plus className="size-3.5" />}
   955→          Создать бронь
   956→        </Button>
   957→      </div>
   958→    </div>
   959→  );
   960→}
